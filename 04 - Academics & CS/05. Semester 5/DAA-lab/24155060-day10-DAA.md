## 1. Fibonacci (Tabulation)


```c
/* q1_fib.c */
#include <stdio.h>
int main(void) {
    int n;
    printf("Enter n: ");
    if (scanf("%d", &n) != 1 || n < 0) { printf("Invalid input\n"); return 0; }
    long long dp[n + 1];
    dp[0] = 0;
    if (n >= 1) dp[1] = 1;
    for (int i = 2; i <= n; i++) dp[i] = dp[i - 1] + dp[i - 2];
    printf("DP table: ");
    for (int i = 0; i <= n; i++) printf("%lld ", dp[i]);
    printf("\nFibonacci number: %lld\n", dp[n]);
    return 0;
}
```

## 2. 0/1 Knapsack (with item reconstruction)

```c
/* q2_knapsack.c */
#include <stdio.h>
#define MAXN 50
#define MAXW 500

int main(void) {
    int n, W, wt[MAXN], pf[MAXN];
    printf("Enter number of items and capacity: ");
    scanf("%d %d", &n, &W);
    printf("Enter weights of %d items: ", n);
    for (int i = 0; i < n; i++) scanf("%d", &wt[i]);
    printf("Enter profits of %d items: ", n);
    for (int i = 0; i < n; i++) scanf("%d", &pf[i]);

    int dp[MAXN + 1][MAXW + 1];
    for (int i = 0; i <= n; i++) {
        for (int j = 0; j <= W; j++) {
            if (i == 0 || j == 0) dp[i][j] = 0;
            else {
                dp[i][j] = dp[i - 1][j];
                if (wt[i - 1] <= j) {
                    int v = dp[i - 1][j - wt[i - 1]] + pf[i - 1];
                    if (v > dp[i][j]) dp[i][j] = v;
                }
            }
        }
    }

    printf("\nDP table:\n      ");
    for (int j = 0; j <= W; j++) printf("%4d", j);
    printf("\n");
    for (int i = 0; i <= n; i++) {
        printf("i=%d : ", i);
        for (int j = 0; j <= W; j++) printf("%4d", dp[i][j]);
        printf("\n");
    }

    int chosen[MAXN], c = 0, i = n, j = W;
    while (i > 0 && j > 0) {
        if (dp[i][j] != dp[i - 1][j]) { chosen[c++] = i; j -= wt[i - 1]; }
        i--;
    }

    int tw = 0;
    for (int k = 0; k < c; k++) tw += wt[chosen[k] - 1];

    printf("\nMaximum profit: %d\n", dp[n][W]);
    printf("Selected items: ");
    for (int k = c - 1; k >= 0; k--) printf("%d ", chosen[k]);
    printf("\nTotal weight: %d\n", tw);
    return 0;
}
```

## 3. Rod Cutting

```c
/* q3_rodcut.c */
#include <stdio.h>
int main(void) {
    int n;
    printf("Enter rod length: ");
    scanf("%d", &n);
    int price[n + 1], dp[n + 1], cut[n + 1];
    printf("Enter prices for lengths 1..%d: ", n);
    for (int i = 1; i <= n; i++) scanf("%d", &price[i]);

    dp[0] = 0;
    for (int j = 1; j <= n; j++) {
        dp[j] = -1; cut[j] = 0;
        for (int i = 1; i <= j; i++) {
            int v = price[i] + dp[j - i];
            if (v > dp[j]) { dp[j] = v; cut[j] = i; }
        }
    }

    printf("\nDP table (best revenue for each rod length):\n");
    for (int j = 0; j <= n; j++) printf("length %d : %d\n", j, dp[j]);

    printf("\nMaximum revenue: %d\n", dp[n]);
    printf("Optimal cuts: ");
    int j = n;
    while (j > 0) { printf("%d ", cut[j]); j -= cut[j]; }
    printf("\n");
    return 0;
}
```

## 4. Minimum Cost Path in a Grid

```c
/* q4_minpath.c */
#include <stdio.h>
int main(void) {
    int m, n;
    printf("Enter number of rows and columns: ");
    scanf("%d %d", &m, &n);
    int g[m][n], dp[m][n];
    printf("Enter the grid (%d x %d):\n", m, n);
    for (int i = 0; i < m; i++)
        for (int j = 0; j < n; j++) scanf("%d", &g[i][j]);

    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            if (i == 0 && j == 0) dp[i][j] = g[i][j];
            else if (i == 0) dp[i][j] = dp[i][j - 1] + g[i][j];
            else if (j == 0) dp[i][j] = dp[i - 1][j] + g[i][j];
            else {
                int up = dp[i - 1][j], left = dp[i][j - 1];
                dp[i][j] = (up < left ? up : left) + g[i][j];
            }
        }
    }

    printf("\nDP cost table:\n");
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) printf("%4d", dp[i][j]);
        printf("\n");
    }

    int pi[m * n], pj[m * n], len = 0, i = m - 1, j = n - 1;
    while (1) {
        pi[len] = i; pj[len] = j; len++;
        if (i == 0 && j == 0) break;
        if (i == 0) j--;
        else if (j == 0) i--;
        else if (dp[i - 1][j] <= dp[i][j - 1]) i--;
        else j--;
    }

    printf("\nMinimum path cost: %d\n", dp[m - 1][n - 1]);
    printf("Path: ");
    for (int k = len - 1; k >= 0; k--) {
        printf("(%d,%d)", pi[k], pj[k]);
        if (k > 0) printf(" -> ");
    }
    printf("\n");
    return 0;
}
```

## 5. Longest Increasing Subsequence

```c
/* q5_lis.c */
#include <stdio.h>
int main(void) {
    int n;
    printf("Enter number of elements: ");
    scanf("%d", &n);
    int a[n];
    printf("Enter the elements: ");
    for (int i = 0; i < n; i++) scanf("%d", &a[i]);

    int dp[n], prev[n];
    for (int i = 0; i < n; i++) { dp[i] = 1; prev[i] = -1; }
    for (int i = 1; i < n; i++)
        for (int j = 0; j < i; j++)
            if (a[j] < a[i] && dp[j] + 1 >= dp[i]) { dp[i] = dp[j] + 1; prev[i] = j; }

    int best = 0;
    for (int i = 0; i < n; i++) if (dp[i] >= dp[best]) best = i;

    printf("\nDP (LIS length ending at each index):\n");
    for (int i = 0; i < n; i++) printf("  index %d (value %2d) -> %d\n", i, a[i], dp[i]);

    int seq[n], c = 0;
    for (int i = best; i != -1; i = prev[i]) seq[c++] = a[i];

    printf("\nLength of LIS: %d\n", dp[best]);
    printf("One LIS: ");
    for (int i = c - 1; i >= 0; i--) printf("%d ", seq[i]);
    printf("\n");
    return 0;
}
```

## 6. Coin Change — Minimum Number of Coins

```c
/* q6_coinchange.c */
#include <stdio.h>
#define INF 1000000
int main(void) {
    int k;
    printf("Enter number of denominations: ");
    scanf("%d", &k);
    int c[k];
    printf("Enter the denominations: ");
    for (int i = 0; i < k; i++) scanf("%d", &c[i]);
    int V;
    printf("Enter the target amount: ");
    scanf("%d", &V);

    int dp[V + 1];
    dp[0] = 0;
    for (int j = 1; j <= V; j++) {
        dp[j] = INF;
        for (int i = 0; i < k; i++)
            if (c[i] <= j && dp[j - c[i]] + 1 < dp[j]) dp[j] = dp[j - c[i]] + 1;
    }

    printf("\nDP table (min coins for 0..%d):\n", V);
    for (int j = 0; j <= V; j++) printf("%d ", dp[j]);
    printf("\n\n");

    if (dp[V] >= INF) { printf("Amount %d cannot be formed.\n", V); return 0; }

    for (int i = 0; i < k - 1; i++)
        for (int j = i + 1; j < k; j++)
            if (c[j] > c[i]) { int t = c[i]; c[i] = c[j]; c[j] = t; }

    printf("Minimum number of coins: %d\n", dp[V]);
    printf("Coins used: ");
    int v = V;
    while (v > 0) {
        for (int i = 0; i < k; i++)
            if (c[i] <= v && dp[v] == dp[v - c[i]] + 1) {
                printf("%d ", c[i]); v -= c[i]; break;
            }
    }
    printf("\n");
    return 0;
}
```

## 7. Edit Distance

```c
/* q7_editdist.c */
#include <stdio.h>
#include <string.h>
#define MAX 200

int main(void) {
    char a[MAX], b[MAX];
    printf("Enter first string: ");
    scanf("%s", a);
    printf("Enter second string: ");
    scanf("%s", b);

    int m = strlen(a), n = strlen(b);
    int dp[m + 1][n + 1];

    for (int i = 0; i <= m; i++) dp[i][0] = i;
    for (int j = 0; j <= n; j++) dp[0][j] = j;

    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            if (a[i - 1] == b[j - 1]) dp[i][j] = dp[i - 1][j - 1];
            else {
                int mn = dp[i - 1][j];
                if (dp[i][j - 1] < mn) mn = dp[i][j - 1];
                if (dp[i - 1][j - 1] < mn) mn = dp[i - 1][j - 1];
                dp[i][j] = 1 + mn;
            }
        }
    }

    printf("\nDP table:\n      ");
    for (int j = 0; j <= n; j++) printf("%4c", j ? b[j - 1] : '"');
    printf("\n");
    for (int i = 0; i <= m; i++) {
        printf("%4c: ", i ? a[i - 1] : '"');
        for (int j = 0; j <= n; j++) printf("%4d", dp[i][j]);
        printf("\n");
    }

    printf("\nMinimum edit distance: %d\n", dp[m][n]);
    return 0;
}
```

## 8. Subset Sum (with reconstruction)

```c
/* q8_subset.c */
#include <stdio.h>
#define MAXN 100
#define MAXS 1000

int main(void) {
    int n, S, a[MAXN];
    printf("Enter number of elements: ");
    scanf("%d", &n);
    printf("Enter the elements: ");
    for (int i = 0; i < n; i++) scanf("%d", &a[i]);
    printf("Enter the target sum: ");
    scanf("%d", &S);

    static int dp[MAXN + 1][MAXS + 1];
    for (int i = 0; i <= n; i++) dp[i][0] = 1;

    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= S; j++) {
            dp[i][j] = dp[i - 1][j];
            if (a[i - 1] <= j && dp[i - 1][j - a[i - 1]]) dp[i][j] = 1;
        }
    }

    printf("\nDP table (rows = elements, cols = sums 0..%d):\n     ", S);
    for (int j = 0; j <= S; j++) printf("%3d", j);
    printf("\n");
    for (int i = 0; i <= n; i++) {
        printf("%2d:  ", i);
        for (int j = 0; j <= S; j++) printf("%3d", dp[i][j]);
        printf("\n");
    }

    if (!dp[n][S]) { printf("\nNo subset with target sum %d exists.\n", S); return 0; }

    int pick[MAXN], c = 0, i = n, j = S;
    while (i > 0 && j > 0) {
        if (dp[i - 1][j]) i--;
        else { pick[c++] = a[i - 1]; j -= a[i - 1]; i--; }
    }

    printf("\nSubset with target sum exists.\n");
    printf("Subset: ");
    for (int k = c - 1; k >= 0; k--) printf("%d ", pick[k]);
    printf("\n");
    return 0;
}
```

## 9. Optimal Binary Search Tree

```c
/* q9_obst.c */
#include <stdio.h>
#define MAXN 50
#define INF 1000000000

int main(void) {
    int n;
    printf("Enter number of keys: ");
    scanf("%d", &n);
    int key[MAXN + 1], f[MAXN + 1];
    printf("Enter sorted keys: ");
    for (int i = 1; i <= n; i++) scanf("%d", &key[i]);
    printf("Enter their frequencies: ");
    for (int i = 1; i <= n; i++) scanf("%d", &f[i]);

    static int cost[MAXN + 2][MAXN + 2], root[MAXN + 2][MAXN + 2], sum[MAXN + 2][MAXN + 2];

    for (int i = 1; i <= n; i++) {
        cost[i][i] = f[i];
        sum[i][i] = f[i];
        root[i][i] = i;
    }

    for (int len = 2; len <= n; len++) {
        for (int i = 1; i + len - 1 <= n; i++) {
            int j = i + len - 1;
            sum[i][j] = sum[i][j - 1] + f[j];
            cost[i][j] = INF;
            for (int r = i; r <= j; r++) {
                int c = (r > i ? cost[i][r - 1] : 0)
                      + (r < j ? cost[r + 1][j] : 0)
                      + sum[i][j];
                if (c < cost[i][j]) { cost[i][j] = c; root[i][j] = r; }
            }
        }
    }

    printf("\nCost table (rows = start index, cols = end index):\n        ");
    for (int j = 1; j <= n; j++) printf("j=%d   ", j);
    printf("\n");
    for (int i = 1; i <= n; i++) {
        printf("i=%d   ", i);
        for (int j = 1; j <= n; j++) {
            if (i > j) printf("%6s", "-");
            else printf("%6d", cost[i][j]);
        }
        printf("\n");
    }

    printf("\nMinimum search cost: %d\n", cost[1][n]);
    printf("Root key: %d\n", key[root[1][n]]);
    return 0;
}
```

## 10. All-Pairs Shortest Paths (Floyd–Warshall)

```c
/* q10_floyd.c */
#include <stdio.h>
#define MAXV 50
#define NOEDGE 9999

int main(void) {
    int V;
    printf("Enter number of vertices: ");
    scanf("%d", &V);
    int d[MAXV][MAXV];
    printf("Enter the adjacency matrix (%d = no edge):\n", NOEDGE);
    for (int i = 0; i < V; i++)
        for (int j = 0; j < V; j++) scanf("%d", &d[i][j]);

    for (int k = 0; k < V; k++)
        for (int i = 0; i < V; i++)
            for (int j = 0; j < V; j++)
                if (d[i][k] != NOEDGE && d[k][j] != NOEDGE &&
                    d[i][k] + d[k][j] < d[i][j])
                    d[i][j] = d[i][k] + d[k][j];

    printf("\nShortest distance matrix:\n");
    for (int i = 0; i < V; i++) {
        for (int j = 0; j < V; j++) printf("%d ", d[i][j]);
        printf("\n");
    }
    return 0;
}
```
